\begin{equation}
\sin \alpha = \frac{2 \tan \frac{\alpha}{2}}{1+\tan^2 \frac{\alpha}{2}} \label{eq:trig1}
\end{equation}

\begin{equation}
\cos \alpha = \frac{1- \tan^2 \frac{\alpha}{2}}{1+\tan^2 \frac{\alpha}{2}} \label{eq:trig2}
\end{equation}

\begin{equation}
\tan \alpha = \frac{2 \tan \frac{\alpha}{2}}{1-\tan^2 \frac{\alpha}{2}} \label{eq:trig3}
\end{equation}
