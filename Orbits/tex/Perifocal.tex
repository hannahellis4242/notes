\section{Introduction}

In this chapter we will introduce the perifocal coordinate system and 
derive the equations of motion in this coordinate system. We will also 
establish the angular momentum, and total energy. Finally we will show
that closed solutions to the equations of motion in the perifocal coordinate
system are ellipses.

As a bonus, we will also revisit the eccentricity vector, and show how it
relates to the solutions from this chapter.

\section{Plane Polar Coordinates}
In this section we will derive the equations of motion for an object acted upon by gravity in plane
polar coordinates. To do this we will first consider plane polar coordinates themselves and the derivatives of the position vector within the coordinate system.
\begin{figure}[h]
  \centering
\includegraphics[width=0.6\textwidth]{./img/perifocal.png} 
\caption{Position and unit vectors in plane polar coordinates} \label{fig:PolarCoordinates}
\end{figure}

\subsection{Polar unit vectors}
The unit vectors of plane polar coordinates can be determined from the Cartesian unit vectors as
\begin{align}
\hat{\mathbf{r}} &= \uveci \cos( \nu ) + \uvecj \sin( \nu ) \label{eq:unitrad} \\
\hat{\mathbf{\nu}}  &= - \uveci \sin( \nu ) + \uvecj \cos( \nu ) \label{eq:unitangle}
\end{align}
Where $r$ is the length of the position vector, and $\nu$ is the angle between $\uveci$ and $\mathbf{r}$.
Since the unit vectors $\hat{\mathbf{r}}$ and $\hat{\mathbf{\nu}}$ depend on $\nu$ which changes with time, the unit vectors also change with time. We will begin by looking at the derivative of $\hat{\mathbf{r}}$ with respect to time.

\begin{equation} \label{eq:radvel}
\frac{d {\hat{\mathbf{r}}}}{d t} = \frac{ d \nu}{d t}\frac{d}{d \nu} \hat{\mathbf{r}} =  \frac{ d \nu}{d t} \left[ -\uveci \sin( \nu ) + \uvecj \cos( \nu ) \right] = \frac{ d \nu}{d t} \hat{\mathbf{\nu}}
\end{equation}

where we have used equation \ref{eq:unitangle} to replace the typical Cartesian coordinate unit vectors.

\begin{equation}\label{eq:angvel}
\frac{d {\hat{\mathbf{\nu}}}}{d t}=\frac{d \nu}{d t} \frac{d}{dt}{\hat{\mathbf{\nu}}}=\frac{d \nu}{d t}  \left[ - \uveci \cos( \nu ) - \uvecj \sin( \nu ) \right]=- \frac{d \nu}{d t} \hat{\mathbf{r}}
\end{equation}

 Since $\frac{d {\hat{\mathbf{r}}}}{d t}$ too depends on $\nu$ so is also time dependent.

\begin{equation} \label{eq:radaccel}
\frac{d^2 {\hat{\mathbf{r}}}}{d t^2} = \frac{d}{dt}\left[ \frac{ d \nu}{d t} \hat{\mathbf{\nu}} \right]=\frac{d^2 \nu}{d t^2}\hat{\mathbf{\nu}}+\frac{d \nu}{d t} \frac{d \hat{\mathbf{\nu}}}{dt}=\frac{d^2 \nu}{d t^2}\hat{\mathbf{\nu}}-\left(\frac{d \nu}{d t} \right)^2  \hat{\mathbf{r}} 
\end{equation}

\subsection{Velocity and Acceleration}
Here we will calculate the acceleration in terms of the polar unit vectors. 

\subsubsection{Position}
Starting with the position vector
\begin{equation}
\textbf{r}= r \hat{\textbf{r}}
\end{equation}
This is the length of the position vector in the direction of $\hat{\mathbf{r}}$.

\subsubsection{Velocity}
The first derivative will be the velocity of the position vector in terms of the polar unit vectors
\begin{equation}\label{eq:vel}
\frac{d \mathbf{r}}{d t} = \frac{d r}{d t} \hat{\mathbf{r}} + r \frac{d \hat{\mathbf{r}}}{d t}=\frac{d r}{d t} \hat{\mathbf{r}} + r \omega \hat{\mathbf{\nu}}
\end{equation}
The velocity is the sum of the radial velocity in the radial direction $\dot{r}$ and the angular part of the velocity $r \omega$ in the angular direction.

\subsubsection{Acceleration}

\begin{align}\label{eq:acceleration}
\frac{ d^2 \mathbf{r}}{ d t^2} &= \frac{ d^2 r}{d t^2} \hat{\mathbf{r}} + \frac{d r}{d t} \frac{d \hat{\mathbf{r}}}{d t} +\frac{d r}{d t} \omega \hat{\mathbf{\nu}} +r \frac{d \omega}{d t} \hat{\mathbf{\nu}}+r \omega \frac{d \hat{\mathbf{\nu}}}{d t} \nonumber \\
&=  \frac{ d^2 r}{d t^2} \hat{\mathbf{r}} + \frac{d r}{d t} \omega \hat{\mathbf{\nu}} +\frac{d r}{d t} \omega \hat{\mathbf{\nu}} +r \frac{d \omega}{d t} \hat{\mathbf{\nu}}-r \omega^2 \hat{\mathbf{r}} \nonumber \\
&=\left[  \frac{ d^2 r}{d t^2} -r \omega^2 \right] \hat{\mathbf{r}} + \left[ 2 \frac{d r}{d t} \omega + r \frac{ d \omega}{d t} \right] \hat{\mathbf{\nu}}
\end{align}

\section{Gravity}
\subsection{Gravity as a central force}
The force between two particles a distance $r$ apart with mass $M$ and $m$ respectivly is given by
\begin{equation}
\mathbf{F}=-\frac{G M m}{r^2} \hat{\mathbf{r}}
\end{equation}

The acceleration on the particle of mass $m$ due to the force of gravity between the two particles is
\begin{equation}
\ddot{\mathbf{r}} =\frac{ \mathbf{F} }{m}=-\frac{GM}{r^2} \hat{\mathbf{r}}
\end{equation}
Using equation \ref{eq:acceleration} we can split the acceleration into radial and angular parts
\begin{align}
-\frac{GM}{r^2} &=\ddot{r} - r \omega^2 \label{eq:radialAcceleration} \\
0 &=r \ddot{\nu} + 2 \dot{r} \dot{\nu} \label{eq:angularAcceleration}
\end{align}

\section{Angular Momentum}
The angular momentum of a particle is given by
\begin{align}
\mathbf{L}&=\mathbf{r} \times \mathbf{p} \nonumber \\
&=m \mathbf{r}\times \frac{d \mathbf{r}}{d t} \nonumber \\
&=m r \hat{\mathbf{r}} \times \left[ \dot{r} \hat{\mathbf{r}} + r \omega \hat{\mathbf{\nu}} \right] \nonumber \\
&=m \left[ r \dot{r}\hat{\mathbf{r}} \times \hat{\mathbf{r}} + r^2 \omega \hat{\mathbf{r}} \times\hat{\mathbf{\nu}} \right] \nonumber \\
&=m r^2 \omega \uveck \label{eq:AngularMomentum}
\end{align}

We can also write this as $\mathbf{L} = l \uveck$ were
\begin{equation}
l=mr^2\omega \label{eq:AngularMomentumScalar}
\end{equation}

\subsection{Conservation Of Angular Momentum}
We can show now that angular momentum is conserved under a central force by taking the derivative with respect to time.

\begin{equation}
\frac{d\mathbf{L}}{dt} = \frac{d l \uveck}{dt} = \frac{d l}{dt}\uveck
\end{equation}
\begin{align}
\frac{dl}{dt}&=m\frac{d}{dt}\left[r^2\omega\right]\nonumber \\
&=m\left[\frac{dr^2}{dt}\omega + r^2\frac{d\omega}{dt}\right]\nonumber \\
&=m\left[\frac{dr^2}{dr}\frac{dr}{dt}\omega + r^2\frac{d\omega}{dt}\right] \nonumber \\
&=m\left[2r\frac{dr}{dt}\omega + r^2\frac{d\omega}{dt}\right] \nonumber \\
&=mr\left[2\frac{dr}{dt}\omega + r\frac{d\omega}{dt}\right]
\end{align}
The part inside the brackets is the angular acceleration which for a central force is zero\footnote{see equation \ref{eq:angularAcceleration}} and so
\begin{align}
\frac{dl}{dt}&=mr \times 0 \nonumber \\
&=0
\end{align}
Which means that the angular momentum is constant as seen before.

\section{Energy}
\subsection{Potential Energy}
The potential energy is the work done in moving the particle from a radius of $\infty$ to $r$.
So the work done is given by
\begin{align}
U(r)&=\int_{\infty}^{r} \mathbf{F}(r') \cdot d\mathbf{r}'  \nonumber \\
U(r)&=\int_{\infty}^{r} -\frac{GMm}{r'^2} \hat{\mathbf{r}} \cdot dr' \hat{\mathbf{r}}  \nonumber \\
U(r)&=-GMm\int_{\infty}^{r} \frac{dr'}{r'^2} \nonumber \\
U(r)&=-GMm \left[ \frac{-1}{r'} \right]_{\infty}^{r} \nonumber \\
U(r)&=GMm \left[ \frac{1}{\infty} - \frac{1}{r} \right] \nonumber \\
U(r)&=- \frac{GMm}{r} \label{eq:potentialEnergy}
\end{align}

\subsection{Kinetic Energy}
The kinetic energy of a particle is given by
\begin{equation}
T(\dot{r})=\frac{1}{2}m\dot{\mathbf{r}}\cdot\dot{\mathbf{r}} \label{eq:KineticEnergy}
\end{equation}
using eqation \ref{eq:vel} gives 
\begin{align}
T(\dot{r})&=\frac{1}{2}m\left[ \dot{r}\hat{\mathbf{r}}+r\omega\hat{\mathbf{\nu}}\right]\cdot\left[ \dot{r}\hat{\mathbf{r}}+r\omega\hat{\mathbf{\nu}}\right] \nonumber \\
&=\frac{1}{2}m \dot{r}^2 + \frac{1}{2}mr^2 \omega^2 \label{eq:Gravity:KineticEnergy}
\end{align}

we can also make the subsitution using using the magnitude of the angular momentum\footnote{equation \ref{eq:AngularMomentumScalar}} $l=mr^2\omega$

\begin{align}
l&=mr^2\omega \nonumber \\
\omega&=\frac{l}{mr^2} \label{eq:Gravity:AngularMomentum}
\end{align}

using equation \ref{eq:Gravity:AngularMomentum} in equation \ref{eq:Gravity:KineticEnergy} leads to

\begin{align}
T(\dot{r})&=\frac{1}{2}m \dot{r}^2 + \frac{1}{2}mr^2 \omega^2 \nonumber \\
&=\frac{1}{2}m \dot{r}^2 + \frac{1}{2}mr^2 \frac{l^2}{m^2r^4} \nonumber \\
&=\frac{1}{2}m \dot{r}^2 + \frac{l^2}{2mr^2} \label{eq:Gravity:KineticEnergyFinal}
\end{align}

\subsubsection{Total Energy}
Finally the total energy can be found by putting together the equation for kinetic energy and potential energy
\begin{align}
E(r,\dot{r})=T(\dot{r})+U(r) \nonumber \\
E(r,\dot{r})=\frac{1}{2}m \dot{r}^2 + \frac{l^2}{2mr^2}- \frac{GMm}{r}\label{eq:Gravity:TotalEnergy}
\end{align}

sometimes the last two terms of equation \ref{eq:Gravity:TotalEnergy} are put together to form a new potential energy type term that is dependent only on the radius and the angular momentum.

\subsection{Conservation of Total Energy}

Just like with angular momentum we can show that total energy is conserved by taking the derivative with respect to time.

\begin{align}
\frac{dE}{dt}&=\frac{d}{dt}\left[\frac{1}{2}m \dot{r}^2 +  \frac{l^2}{2mr^2}-\frac{GMm}{r}\right] \nonumber \\
&=\frac{1}{2}m \frac{d\dot{r}^2}{dt} +  \frac{l^2}{2m}\frac{dr^{-2}}{dt}-GMm\frac{dr^{-1}}{dt} \nonumber \\
&=\frac{1}{2}m \frac{d\dot{r}^2}{d\dot{r}}\frac{d\dot{r}}{dt} +  \frac{l^2}{2m}\frac{dr^{-2}}{dr}\frac{dr}{dt}-GMm\frac{dr^{-1}}{dr}\frac{dr}{dt} \nonumber \\
&=\frac{1}{2}m (2\dot{r})\ddot{r}+ \frac{l^2}{2m}\left(\frac{-2}{r^3}\right)\dot{r}+\frac{GMm}{r^2}\dot{r} \nonumber \\
&=m \dot{r}\ddot{r}- \frac{l^2}{mr^3}\dot{r}+\frac{GMm}{r^2}\dot{r}
\end{align}

using the magnitude of the angular momentum\footnote{equation \ref{eq:AngularMomentumScalar}} $l=mr^2\omega$

\begin{align}
\frac{dE}{dt} &=m \dot{r}\ddot{r}- \frac{l^2}{mr^3}\dot{r}+\frac{GMm}{r^2}\dot{r} \nonumber \\
&=m \dot{r}\ddot{r}- \frac{m^2r^4\omega^2}{mr^3}\dot{r}+\frac{GMm}{r^2}\dot{r} \nonumber \\
&=m \dot{r}\ddot{r}-mr\omega^2\dot{r}+\frac{GMm}{r^2}\dot{r} \nonumber \\
&=m \dot{r}\left[\ddot{r}-r\omega^2+\frac{GM}{r^2}\right]
\end{align}

using the radial part of the acceleration\footnote{equation \ref{eq:radialAcceleration}} $-\frac{GM}{r^2}= \ddot{r} -r \omega^2$ we get
\begin{align}
\frac{dE}{dt}&=m \dot{r}\left[-\frac{GM}{r^2}+\frac{GM}{r^2}\right] \nonumber \\
&=m \dot{r}\left[0\right] \nonumber \\
&=0
\end{align}

This then shows that the total energy does not change with respect to time, so is a constant and also conserved.

\section{Eliptical Orbits}
We can show that a general solution to the equations of motion is an elipse.

\subsection{Change of variables}
Firstly we start by letting $r=\frac{1}{u}$, so that

\begin{align}
\frac{dr}{dt} &= \frac{du^{-1}}{dt}\nonumber \\
&= \frac{du^{-1}}{du}\frac{du}{dt}\nonumber \\
&= -\frac{1}{u^2}\frac{du}{dt}\nonumber \\
&= -r^2\frac{du}{dt}
\end{align}

next we introduce an $\omega$ by chain rule

\begin{align}
\frac{dr}{dt}
&= -r^2\frac{du}{dt}\nonumber \\
\frac{dr}{dt} &= -r^2\frac{du}{d\nu}\frac{d\nu}{dt} \nonumber \\
&= -r^2\omega\frac{du}{d\nu} \nonumber
\end{align}

And using the magnitude of angular momentum\footnote{equation \ref{eq:AngularMomentumScalar}} $l=mr^2\omega$ or rearranging $\omega = \frac{l}{mr^2}$

\begin{align}
\frac{dr}{dt}&= -r^2\omega\frac{du}{d\nu} \nonumber \\
\frac{dr}{dt}&= -r^2\frac{l}{mr^2}\frac{du}{d\nu} \nonumber \\
\frac{dr}{dt}&= -\frac{l}{m}\frac{du}{d\nu} \nonumber \\
\frac{dr}{dt}&= -\frac{l}{m}\frac{du}{d\nu}
\end{align}

Differentiating again with respect to time gives

\begin{align}
\frac{dr^2}{dt^2}&= -\frac{l}{m}\frac{d}{dt}\frac{du}{d\nu} \nonumber \\
&= -\frac{l}{m}\frac{d\nu}{dt}\frac{d}{d\nu}\frac{du}{d\nu}\nonumber \\
&= -\frac{l}{m}\omega\frac{d^2u}{d\nu^2}
\end{align}

We can substitute the magnitude of angular momentum\footnote{equation \ref{eq:AngularMomentumScalar}} $l=mr^2\omega$ again to get

\begin{align}
\frac{dr^2}{dt^2}&= -\frac{l}{m}\omega\frac{d^2u}{d\nu^2} \nonumber \\
&= -\frac{mr^2\omega}{m}\omega\frac{d^2u}{d\nu^2} \nonumber \\
&= -r^2\omega^2\frac{d^2u}{d\nu^2}
\end{align}

We can now substitute this into the radial part of the acceleration\footnote{equation \ref{eq:radialAcceleration}}

\begin{align}
-\frac{GM}{r^2}&= \frac{ d^2 r}{d t^2} -r \omega^2  \nonumber \\
&=-r^2\omega^2\frac{d^2u}{d\nu^2} -r\omega^2 \nonumber \\
&=-r^2\omega^2\left[\frac{d^2u}{d\nu^2}+\frac{1}{r}\right] \nonumber \\
GM&=r^4\omega^2\left[\frac{d^2u}{d\nu^2}+\frac{1}{r}\right]
\end{align}

Next we recall that $u=\frac{1}{r}$ and that $\frac{l}{m}=r^2\omega$ and substitute these in

\begin{align}
GM&=\left(\frac{l}{m}\right)^2\left[\frac{d^2u}{d\nu^2}+u\right] \nonumber\\
\frac{d^2u}{d\nu^2}+u &=\frac{GMm^2}{l^2}
\end{align}

To simplify this equation we will use $\alpha=\frac{GMm^2}{l^2}$

\subsection{Solution}

The direct solution to
\begin{equation}
\frac{d^2u}{d\nu^2}+u =\alpha
\end{equation}

is $u=\alpha$

However because there is also a function that results in zero, we must include that

\begin{align}
\frac{d^2u}{d\nu^2}+u &=0 \nonumber \\
\frac{d^2u}{d\nu^2} &=-u
\end{align}

we can take the solution that $u=c\cos\nu$ which gives the general overall solution of

\begin{equation}
u(\nu)=\alpha + c \cos \nu
\end{equation}

We can pick an initial condition when $\nu=0$ the body is at periapsis or $r=p=A(1-e)$ which gives $u=\frac{1}{A(1-e)}$ (see the ellipse notes for $p=A(1-e)$)


\begin{align}
\frac{1}{A(1-e)}&=\alpha+c \nonumber \\
\alpha&=\frac{1}{A(1-e)} - c
\end{align}


we could similarly hae picked the condition that when $\nu=\pi$ the body is at apoapsis or $r=a=A(1+e)$ which gives $u=\frac{1}{A(1+e)}$ (see ellipse notes for $a=A(1+e)$)


\begin{align}
\frac{1}{A(1+e)}&=\alpha-c \nonumber \\
\alpha&=\frac{1}{A(1+e)} + c
\end{align}

we can equate our two initial conditions to get $c$ in terms of just $a$ and $p$

\begin{align}
\alpha=\frac{1}{A(1-e)}-c&=\frac{1}{A(1+e)} + c \nonumber \\
\frac{1}{A(1-e)}-\frac{1}{A(1+e)}&=2c \nonumber \\
\frac{A(1+e)-A(1-e)}{A^2(1+e)(1-e)}&=2c \nonumber \\
\frac{A+eA-A+eA}{A^2(1-e^2)}&=2c \nonumber \\
\frac{2eA}{A^2(1-e^2)}&=2c \nonumber \\
\frac{e}{A(1-e^2)}&=c
\end{align}

Now that we have $c$ we can use this to determine $\alpha$ in terms of $A$ and $e$

\begin{align}
\alpha&=\frac{1}{A(1+e)} + c \nonumber \\
&=\frac{1}{A(1+e)} + \frac{e}{A(1-e)^2}\nonumber \\
&=\frac{(1-e)}{A(1+e)(1-e)} + \frac{e}{A(1-e)^2}\nonumber \\
&=\frac{1-e+e}{A(1-e^2)} \nonumber \\
&=\frac{1}{A(1-e^2)} \nonumber
\end{align}

So feeding all this back in we have

\begin{align}
u(\nu)&=\alpha + c \cos \nu \nonumber \\
&=\frac{1}{A(1-e^2)} + \frac{e}{A(1-e^2)} \cos \nu \nonumber \\
&=\frac{1+e\cos\nu}{A(1-e^2)}
\end{align}

Now we have a simple form for $u$ we can recall that we chose $u=\frac{1}{r}$, so
\begin{equation}
r=\frac{1+e\cos\nu}{A(1-e^2)}
\end{equation}
which is the equation for a ellipse\footnote{equation \ref{eq:PolarElipse}}.

We also get the interesting result here that

\begin{align}
\alpha = \frac{GMm^2}{l^2}&=\frac{1}{A(1-e^2)} \nonumber \\
A(1-e^2)&=\frac{l^2}{GMm^2} \label{eq:link}
\end{align}
which connects the semi major axis and the eccentricity to physical properties, like the masses and the angular momentum. This will help us to work out the geometry from the physics.
