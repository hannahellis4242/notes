\subsection{Triple Vector Product}

\begin{equation}
\mathbf{a} \times (\mathbf{b} \times \mathbf{c}) = (\mathbf{a} \cdot \mathbf{c})\mathbf{b} -(\mathbf{a} \cdot \mathbf{b} )\mathbf{c} \label{eq:vec1}
\end{equation}

\subsection{Scalar Triple Product}

\begin{equation}
\mathbf{a}\cdot(\mathbf{b}\times\mathbf{c}) = \mathbf{b}\cdot(\mathbf{c}\times\mathbf{a}) \label{eq:vec2}
\end{equation}
